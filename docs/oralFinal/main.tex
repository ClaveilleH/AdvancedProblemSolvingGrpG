\begin{frame}{Trace d'exécution : gains initiaux}{Jade}
  \begin{block}{Gain d'une vidéo dans un cache}
    Somme, sur les endpoints reliés, de (latence actuelle $-$ latence via le cache) $\times$ requêtes.\\
    Exemple : gain(\vb3, cache 2) $= (1000-500)\times 200 + (1000-100)\times 300 = 370\,000$
  \end{block}
  \vspace{0.5em}
  \centering
  \begin{tabular}{ccccc}
    \hline
    \rowcolor{accent!15} & \vb1 & \vb2 & \vb3 & \textbf{Total} \\ \hline
    Cache 1 & 0 & 270\,000 & 180\,000 & 450\,000 \\ \hline
    \rowcolor{orange!25} Cache 2 & 90\,000 & 150\,000 & 370\,000 & \textbf{610\,000} \\ \hline
    Cache 3 & 70\,000 & 0 & 210\,000 & 280\,000 \\ \hline
  \end{tabular}

  \vspace{0.8em}
  Ordre de traitement : \textbf{cache 2}, puis cache 1, puis cache 3.

\begin{frame}{Lecture et structures de données}{Hugo}
  \begin{table}
    \centering
    \small
    \begin{tabular}{llp{5.2cm}}
      \hline
      \rowcolor{accent!15} \textbf{Structure} & \textbf{Contenu} & \textbf{Pourquoi} \\ \hline
      \code{requests} & (vidéo, endpoint, nb) & doublons fusionnés à la lecture : \textbf{$-93\,\%$} de lignes sur les instances 11 et 14 \\ \hline
      \code{adj\_list[v]} & requêtes de la vidéo $v$ & retrouver tout de suite qui demande $v$ \\ \hline
      \code{endpoints[e]} & (lat.\ datacenter, [(cache, lat.)]) & données de l'endpoint $e$ \\ \hline
      \code{caches\_endpoints[c]} & \{endpoint : lat.\} & « cet endpoint est-il relié à $c$ ? » en accès direct \\ \hline
      \code{caches[c]} & \code{set} de vidéos & « $v$ est-elle dans $c$ ? » en temps constant (sac à dos, tabou) \\ \hline
    \end{tabular}
  \end{table}
  \begin{block}{Idée}
    Tout est indexé pour ne jamais parcourir toutes les requêtes quand une seule vidéo change.
  \end{block}
\end{frame}

\begin{frame}{Le sac à dos en pratique}{Hugo}
  \begin{columns}[T]
    \column{0.49\textwidth}
    \begin{block}{Deux programmations dynamiques}
      \begin{itemize}
        \item indexée sur le poids : $n \times$ capacité
        \item indexée sur la valeur : $n \times \sum$ gains
      \end{itemize}
      On prend la plus petite table.
    \end{block}
    \setbeamercolor{block title}{fg=white,bg=red!80!black}
    \begin{block}{Garde-fou}
      Les deux tables dépassent $10^8$ cases $\Rightarrow$ glouton par ratio gain / taille
      (jamais déclenché sur le concours).
    \end{block}
    \column{0.49\textwidth}
    \setbeamercolor{block title}{fg=white,bg=green}
    \begin{block}{Gains mis à jour, pas recalculés}
      Gain de chaque vidéo par cache et meilleure latence par (endpoint, vidéo) :
      vider ou remplir un cache ne revoit que les requêtes de ses vidéos.
    \end{block}
    \begin{block}{Division par le pgcd}
      Tailles et gains divisés par leur pgcd (exact) :
      \begin{itemize}
        \item instance06 : sans lui, tout passe au glouton, score $-35\,\%$
        \item instance03 : $\times 5$ plus rapide
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Recherche locale : évaluer un mouvement}{Hugo}
  \begin{columns}[c]
    \column{0.48\textwidth}
    Mouvement : \textbf{ajouter} ou \textbf{retirer} une vidéo $v$ d'un cache.

    \vspace{0.5em}
    \begin{block}{Delta}
      Seules les requêtes de $v$ peuvent changer :\\[0.3em]
      $\Delta = \text{coût}_v(\text{après}) - \text{coût}_v(\text{avant})$\\[0.3em]
      en parcourant \code{adj\_list[v]} seulement.
    \end{block}
    \begin{itemize}
      \item On restaure la meilleure solution rencontrée
    \end{itemize}
    \column{0.5\textwidth}
    \centering
    \small
    \begin{tabular}{lrrr}
      \hline
      \rowcolor{accent!15} \textbf{Instance} & \textbf{Tout} & \textbf{Delta} & \textbf{Gain} \\ \hline
      instance10 & 324 ms & 0,068 ms & $\times 4\,800$ \\
      instance08 & 263 ms & 0,093 ms & $\times 2\,800$ \\
      instance05 & 838 ms & 0,70 ms & $\times 1\,200$ \\ \hline
    \end{tabular}

    \vspace{0.8em}
    \begin{block}{Une itération sur instance10}
      $100 \times 100$ mouvements, 2 évaluations chacun :\\
      \textbf{$\approx$ 1,4 s} avec le delta, \textbf{$\approx$ 1 h 50} sans.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Limites}{Hugo}
  \begin{itemize}
    \item Python pur : lent sur les grosses instances
    \item La DP garde toute sa table en mémoire, pour retrouver les vidéos choisies
    \item instance12 : $\approx 8 \cdot 10^9$ cases de DP par passe, hors de portée en Python pur (le garde-fou ne se déclenche pas)
    \item Recherche locale : voisinage borné à 100 caches $\times$ 100 vidéos
  \end{itemize}
\end{frame}
